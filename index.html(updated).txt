    // Live Clock Function
    function updateClock() {
      const now = new Date();
      const options = { 
        year: 'numeric', month: 'short', day: 'numeric', 
        hour: '2-digit', minute: '2-digit', second: '2-digit' 
      };
      document.getElementById('live-clock').innerText = now.toLocaleDateString('en-US', options);
    }
    setInterval(updateClock, 1000);
    updateClock();

    // Sidebar Toggle Function
    function toggleComplaintSidebar() {
      document.getElementById('complaintSidebar').classList.toggle('active');
      document.getElementById('overlay').classList.toggle('active');
    }

    let lastLoggedStatus = "Safe";

    // ESP32 Data Fetch & Automated Log Trigger
    async function fetchGasData() {
         try { 
              const response = await fetch('/api/data');
              const data = await response.json();
              
              // Updates your main dashboard table
              document.getElementById('sensor-val').innerText = data.sensor_value;
              document.getElementById('gas-status').innerText = data.gas_status;

              // Automatically generate log card if status moves away from Safe
              if (data.gas_status !== "Safe" && data.gas_status !== lastLoggedStatus) {
                  // Passed data.message right into the function here!
                  appendAutomatedLog(data.gas_status, data.sensor_value, data.message);
                  lastLoggedStatus = data.gas_status; 
              } else if (data.gas_status === "Safe") {
                  lastLoggedStatus = "Safe"; 
              }
        } catch (error) {
            console.error("Error fetching data:", error);
        }
    }

    // Function to dynamically insert new log cards into the sidebar
    function appendAutomatedLog(gas_status, sensor_value, message) {
        const time = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
        const logBox = document.createElement('div');
        logBox.className = 'complaint-card';

        let badgeClass = 'danger';
        // Fixed variable name from 'status' to 'gas_status'
        if (gas_status.includes('Borderline') || gas_status.includes('Warning')) {
            badgeClass = 'warning';
        }

        logBox.innerHTML = `
          <div class="card-header">
            <span class="badge ${badgeClass}">${gas_status}</span>
            <span class="timestamp">${time}</span>
          </div>
          <p class="card-author"><strong>Sensor Value:</strong> ${sensor_value} PPM</p>
          <p class="card-text">${message}</p>
        `;

        const container = document.getElementById('complaintsContainer');
        container.insertBefore(logBox, container.firstChild);
    }

    setInterval(fetchGasData, 1000);
